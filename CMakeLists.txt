cmake_minimum_required(VERSION 3.18)
project(vvtk_fastcsv VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

find_package(Threads REQUIRED)
include(CheckIPOSupported)

option(FASTCSV_USE_FAST_FLOAT "Use fast_float for floating-point parsing (system package or vendored header)" ON)
option(FASTCSV_ENABLE_SIMD "Enable runtime AVX2/SSE2 scanner dispatch on supported x86 compilers" ON)
option(FASTCSV_ENABLE_IPO "Enable interprocedural optimization/LTO when supported" ON)
option(FASTCSV_NATIVE_ARCH "Compile non-dispatched code with -march=native (less portable wheels)" OFF)
option(FASTCSV_BUILD_PYTHON "Build the Python extension" ON)
option(FASTCSV_BUILD_CORE_TESTS "Build the standalone C++ core smoke test" OFF)

add_library(fastcsv_core STATIC
    cpp/fastcsv/reader.cpp
    cpp/fastcsv/writer.cpp
)
set_target_properties(fastcsv_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(fastcsv_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/cpp)
target_compile_features(fastcsv_core PUBLIC cxx_std_20)
target_link_libraries(fastcsv_core PUBLIC Threads::Threads)

if(NOT FASTCSV_ENABLE_SIMD)
  target_compile_definitions(fastcsv_core PRIVATE FASTCSV_DISABLE_SIMD=1)
endif()

if(FASTCSV_USE_FAST_FLOAT)
  find_package(fast_float CONFIG QUIET)
  if(fast_float_FOUND)
    target_link_libraries(fastcsv_core PUBLIC fast_float::fast_float)
    message(STATUS "vvtk_fastcsv: using system fast_float")
  else()
    target_include_directories(fastcsv_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/third_party)
    message(STATUS "vvtk_fastcsv: using vendored fast_float header")
  endif()
  target_compile_definitions(fastcsv_core PUBLIC FASTCSV_HAS_FAST_FLOAT=1)
else()
  message(STATUS "vvtk_fastcsv: fast_float disabled; using std::from_chars")
endif()

if(MSVC)
  target_compile_options(fastcsv_core PRIVATE /O2 /W3 /EHsc)
else()
  target_compile_options(fastcsv_core PRIVATE -O3 -Wall -Wextra)
  if(FASTCSV_NATIVE_ARCH)
    target_compile_options(fastcsv_core PRIVATE -march=native)
  endif()
endif()

if(FASTCSV_BUILD_PYTHON)
  find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core cpp/bindings.cpp)
  target_link_libraries(_core PRIVATE fastcsv_core)
  if(MSVC)
    target_compile_options(_core PRIVATE /O2 /EHsc)
  else()
    target_compile_options(_core PRIVATE -O3)
  endif()
  install(TARGETS _core LIBRARY DESTINATION vvtk_fastcsv)
endif()

if(FASTCSV_ENABLE_IPO)
  check_ipo_supported(RESULT ipo_supported OUTPUT ipo_error LANGUAGES CXX)
  if(ipo_supported)
    set_property(TARGET fastcsv_core PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
    if(FASTCSV_BUILD_PYTHON)
      set_property(TARGET _core PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
    endif()
    message(STATUS "vvtk_fastcsv: IPO/LTO enabled")
  else()
    message(STATUS "vvtk_fastcsv: IPO/LTO unavailable: ${ipo_error}")
  endif()
endif()

if(FASTCSV_BUILD_CORE_TESTS)
  add_executable(fastcsv_core_smoke tests/cpp/core_smoke.cpp)
  target_link_libraries(fastcsv_core_smoke PRIVATE fastcsv_core)
  enable_testing()
  add_test(NAME core_smoke COMMAND fastcsv_core_smoke)
endif()
